%% fig-4-arch.tex -- Adreno 830 architecture (used §6)
\begin{figure}[t]
\centering
\begin{tikzpicture}[
    block/.style={rectangle, draw, fill=blue!8, minimum width=2.6cm,
                  minimum height=0.7cm, align=center, font=\footnotesize},
    block-highlight/.style={block, fill=red!10},
    block-arm/.style={block, fill=green!10},
    arrow/.style={-{Latex[length=2mm]}, thick}
]
% Top: CPU cluster
\node[block-arm] (cpu) {Kryo CPU\\8x Cortex (1+3+4)};
% Middle: System cache
\node[block, below=of cpu] (l3) {System Cache\\3 MB};
% Bottom: Adreno GPU
\node[block-highlight, below=of l3, minimum height=1.2cm] (gpu) {Adreno 830\\1536 ALUs @ 1.25 GHz\\~1.7--2.0 TFLOPs FP32};
% Tile / sort unit
\node[block-highlight, right=of gpu, minimum height=1.2cm] (sort) {Tile-sort\\RA-based\\Reuse-and-Update};
% DRAM
\node[block, below=of gpu, minimum width=10cm] (dram) {LPDDR5X\\8.5 Gbps / 77 GB/s peak BW (LPDDR5X-4800 dual-channel)};
\draw[arrow] (cpu) -- (l3);
\draw[arrow] (l3) -- (gpu);
\draw[arrow] (gpu) -- (sort);
\draw[arrow] (gpu) -- (dram);
\draw[arrow] (sort) -- (dram);
\end{tikzpicture}
\caption{Adreno 830 (Snapdragon 8 Elite) block diagram. The Adreno
830 (red-highlighted) is the deployment substrate for the project:
1536 ALUs at 1.25 GHz with a tile-based deferred rasteriser. The
Reuse-and-Update sort accelerator (also red-highlighted) is the
Neo-inspired hardware-software co-design target that the
project's research roadmap (\S8) calls for. Peak bandwidth
77 GB/s is the practical VRAM ceiling for 4DGS scene
streaming.}
\label{fig:arch}
\end{figure}
